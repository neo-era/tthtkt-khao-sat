Cách tạo backend Google Sheets (1 lần)
1. Tạo Google Sheet mới → Tiện ích mở rộng → Apps Script.
2. Dán đoạn mã dưới, deploy dạng Web app (Execute as: Me; Access: Anyone).
3. Copy URL /exec dán vào ô phía trên. 
function doPost(e){
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var rows = JSON.parse(e.postData.contents);
  var H = ["STT","KhuVuc","TenViTri","Duong","SoNha","Phuong","HaTang",
    "ViDo","KinhDo","HTChieuSang","NguonDien_m",
    "DXDen","DXCap","DXTru","DXPhuKien","UuTien","GhiChu","NguoiKS","ThoiGian"];
  if(sh.getLastRow()===0){ sh.appendRow(H); }
  var data = sh.getDataRange().getValues();
  rows.forEach(function(r){
    var row = -1;
    for(var i=1;i<data.length;i++){ if(String(data[i][0])===String(r[0])){ row=i+1; break; } }
    if(row>0){ sh.getRange(row,1,1,r.length).setValues([r]); }
    else { sh.appendRow(r); data.push(r); }
  });
  return ContentService.createTextOutput("OK");
}
